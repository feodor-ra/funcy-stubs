# Deliberate divergences between the stubs and the funcy runtime.
# Every entry is a design decision, not a gap — see the categories below.

# --- funcy's @decorator factories -------------------------------------------
# inspect sees the raw signature with the leading `call` argument, while the
# stubs model the user-facing call pattern (without `call`).
funcy.debug.log_calls
funcy.debug.log_enters
funcy.debug.log_exits
funcy.debug.print_calls
funcy.debug.print_enters
funcy.debug.print_exits
funcy.flow.collecting
funcy.flow.joining
funcy.flow.post_processing
funcy.flow.retry
funcy.flow.wrap_with

# --- memoize / cache with the .skip attribute -------------------------------
# The stubs declare memoize and cache as variables typed by private protocols
# so they can carry .skip (SkipMemory); stubtest sees plain runtime functions.
# memoize's optional function argument is also named _func at runtime but is
# positional-only in the stubs.
funcy.calc.cache
funcy.calc.memoize

# --- EMPTY-sentinel argument shuffling --------------------------------------
# The runtime accepts a reduced arity via an EMPTY sentinel and reshuffles the
# arguments internally; the stubs express the same thing as overload pairs, so
# defaults land on different parameters than in the runtime signature.
funcy.colls.all
funcy.colls.any
funcy.colls.none
funcy.colls.one
funcy.colls.some
funcy.flow.raiser
funcy.funcs.iffy
funcy.seqs.chunks
funcy.seqs.dropwhile
funcy.seqs.lchunks
funcy.seqs.lpartition
funcy.seqs.partition
funcy.seqs.takewhile
funcy.strings.str_join

# --- positional-only markers ------------------------------------------------
# The arity overloads use fabricated parameter names (iter1..iter5), so the
# stubs mark them positional-only even though the runtime technically accepts
# keywords; passing those fake names at runtime would fail anyway.
funcy.funcs.autocurry
funcy.seqs.lmap
funcy.seqs.lmapcat
funcy.seqs.map
funcy.seqs.mapcat

# --- itertools aliases ------------------------------------------------------
# At runtime cat = chain.from_iterable and concat = chain; the stubs model
# them as plain functions returning Iterable, per the project convention.
funcy.seqs.cat
funcy.seqs.concat
